function make_date(dstr,    d,m,y) {
    split(dstr, a, "/")
    d = a[1]
    m = a[2]
    y = a[3]

    # mktime expects: YYYY MM DD HH MM SS
    return int(mktime(y " " m " " d " 00 00 00") / 86400)
}
