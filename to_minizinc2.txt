# Format shift request into a MiniZinc set
function format_shift(s,    i, out) {
    if (s in leave_shifts) {
        return "{" s "}";
    } else {
        out = "{";
        for (i = 1; i <= length(s); i++) {
            out = out substr(s, i, 1);
            if (i < length(s)) out = out ", ";
        }
        out = out "}";
        return out;
    }
}
