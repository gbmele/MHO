# Reset ONLY per-file state
FNR == 1 {
    delete KEEP
    delete DAY
    day_counter = 0
}

NR > 1 {
    gsub(/,/, "", $1)

    # Detect weekday columns
    for (i = 1; i <= NF; i++) {
        if ($i in weekdays) {
            KEEP[i] = 1
            DAY[i] = ++day_counter
        }
    }

    # Process doctor rows
    if ($1 ~ /^"(Doctor|Specialist) - /) {

        sub(/^"(Doctor|Specialist) - /, "", $1)
        sub(/"$/, "", $1)
        sub(/,.*/, "", $1)

        doc = translate_doc($1)

        if (!(doc in seen)) {
            seen[doc] = 1
            order[++count] = doc
        }

        if (doc == "no_doc") {
            NO_DOCS[++nd] = $1
            next
        }

        # BULLETPROOF SHIFT ACCUMULATION
        delete S
        n = 0

        for (j = 2; j <= NF; j++) {
            if (!KEEP[j-1]) continue

            raw = $j
            gsub(/^ +| +$/, "", raw)
            gsub(/"/, "", raw)

      
            if (raw == "") shift = "apoc"
            else 
            if (raw in translate) shift = translate[raw]
            else
            shift = "\n\n" "XX_____________NO_TRANSLATE "  raw  "\n\n"   ## very raw error screen dump

            n++
            S[n] = shift

            # Append across files
            if (doc in shifts)
                shifts[doc] = shifts[doc] "," shift
            else
                shifts[doc] = shift
        }

        # Emit clean comma-joined output
        out = S[1]
        for (k = 2; k <= n; k++) out = out "," S[k]

        ##print out
    }
}

         print doc "," shifts[doc]
    }

function translate_doc(doc) {
    return (doc in DOC) ? DOC[doc] : "no_doc"
}
