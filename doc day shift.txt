# Custom function to reformat dates
function to_yyyymmdd(date_str,    parts) {
    # split breaks "DD/MM/YYYY" into parts[1]="DD", parts[2]="MM", parts[3]="YYYY"
    if (split(date_str, parts, "/") == 3) {
        # Concatenate them back together in the correct order
        return parts[3] parts[2] parts[1]
    }
    return date_str # Return original if string format doesn't match
}
